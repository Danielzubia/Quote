  <script>
    document.getElementById("quoteForm").addEventListener("submit", function (e) {
      e.preventDefault();

      const basePrice = 800;
      let quote = basePrice;

      if (document.getElementById("newLayout").checked) quote += 300;
      if (document.getElementById("includeADA").checked) quote += 200;

      const rangeLow = quote * 0.9;
      const rangeHigh = quote * 1.1;

      document.getElementById("quoteResult").innerText = 
        `Estimated Price: $${Math.round(rangeLow)}–$${Math.round(rangeHigh)}. We'll connect you with a local pro soon!`;

      sendLeadToBackend();
    });

    function sendLeadToBackend() {
      const lead = {
        address: document.getElementById("address").value,
        newLayout: document.getElementById("newLayout").checked,
        includeADA: document.getElementById("includeADA").checked,
        name: document.getElementById("contactName").value,
        email: document.getElementById("email").value,
        phone: document.getElementById("phone").value,
        timestamp: new Date().toISOString()
      };

      console.log("Sending lead:", lead);

    }
